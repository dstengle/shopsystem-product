    18	  4. Relative markdown links resolve to existing files.
    19	  5. Required headings per type are present

    23	  7. `version` present and `Document History` is the last section
    24	     (definition typedef §Required frontmatter, §Required sections 3);
    25	     generated renderings are exempt.
    26	  8. No numbered-decision reference (review-record typedef §Commitment:
    27	     decisions live as changes in the artifacts they affect).
    28	  9. Each received request in `requests/` at the repository root — the
    29	     directory may not exist yet — carries the frontmatter of a received
    30	     ask: `type: request`, `id`, `status` in {recorded, routed, declined,

    38	     basis/ as before; --derive-chain reads .claude/skills/ as before.
    39	 10. Each decision brief in `briefs/` at the repository root — the
    40	     directory may not exist yet; a brief is a file whose `type` is
    41	     `decision-brief`, and the annexes beside it (a brief's linked full

    51	     `basis/processes/` names — `basis/tools/<name>.py`, written in a

    73	  lint_basis.py                     # lint the whole basis tree

    75	                                    # check 10's rules (exit 0 on pass)

    81	                                    # hand-written)
    82	  lint_basis.py --describe          # answer the standard question: write
    83	                                    # this tool's description — DESCRIPTION
    84	                                    # below, an instance of the
    85	                                    # tool-description data type — as JSON

    91	                                    # by compile_tool.py and from nothing

    94	Any other arguments are a usage failure: one line on standard error

   323	def front_matter(path):
   324	    text = path.read_text()
   325	    m = re.match(r"---\n(.*?)\n---\n", text, re.S)
   326	    if not m:
   327	        return None, text
   328	    try:
   329	        return yaml.safe_load(m.group(1)), text
   330	    except yaml.YAMLError:
   331	        return None, text
   332	
   333	
   334	def yaml_blocks(text):
   335	    out = {}
   336	    for fence in re.findall(r"```yaml\n(.*?)```", text, re.S):
   337	        block = yaml.safe_load(fence)
   338	        if isinstance(block, dict):
   339	            out.update(block)
   340	    return out

   343	def collect_refs(node):
   344	    refs = set()
   345	    if isinstance(node, dict):
   346	        for k, v in node.items():
   347	            if k == "$ref":
   348	                refs.add(v)
   349	            else:
   350	                refs |= collect_refs(v)
   351	    elif isinstance(node, list):
   352	        for item in node:
   353	            refs |= collect_refs(item)
   354	    return refs

   357	def registry():
   358	    reg = {}
   359	    for tree in ("artifacts", "types"):
   360	        for path in (BASIS / tree).glob("*.md"):

   362	            if fm and fm.get("defines"):
   363	                reg[fm["defines"]] = path
   364	    return reg

   367	def lint():
   368	    errors = []
   369	    reg = registry()
   370	
   371	    seen_defines = {}
   372	    for path in sorted(BASIS.rglob("*.md")):
   373	        rel = path.relative_to(BASIS)
   374	        fm, text = front_matter(path)
   375	
   376	        # 1. identity base
   377	        if fm is None:
   378	            errors.append(f"{rel}: front-matter missing or unparseable (definition §Required frontmatter)")
   379	            continue
   380	        for field in IDENTITY_BASE:
   381	            if field not in fm:
   382	                errors.append(f"{rel}: front-matter lacks `{field}` (definition §Required frontmatter)")
   383	        if fm.get("type") not in OWNER_EXEMPT and "owner" not in fm:
   384	            errors.append(f"{rel}: front-matter lacks `owner` (definition §Required frontmatter)")
   385	
   386	        # 7. version + Document History (definition §Required frontmatter,
   387	        #    §Required sections 3); generated renderings are exempt.
   388	        if not fm.get("generated"):
   389	            if "version" not in fm:
   390	                errors.append(f"{rel}: front-matter lacks `version` (definition §Required frontmatter)")
   391	            h2s = re.findall(r"^## (.+)$", text, re.M)
   392	            if not h2s or h2s[-1].strip() != "Document History":
   393	                errors.append(f"{rel}: `Document History` is not the last section (definition §Required sections 3)")
   394	            if "verified-by" in fm:
   395	                errors.append(f"{rel}: review-log frontmatter `verified-by` is ruled out (definition §Required sections 3)")
   396	
   397	        # 2. defines uniqueness
   398	        if fm.get("defines"):
   399	            if fm["defines"] in seen_defines:
   400	                errors.append(f"{rel}: duplicate defines `{fm['defines']}` (also {seen_defines[fm['defines']]})")
   401	            seen_defines[fm["defines"]] = rel
   402	
   403	        # 4. relative links resolve
   404	        for target in re.findall(r"\]\(([^)#\s]+)\)", text):
   405	            if target.startswith(("http://", "https://", "pkg:")):
   406	                continue
   407	            if not (path.parent / target).exists():
   408	                errors.append(f"{rel}: broken link `{target}`")
   409	
   410	        # 5. required headings
   411	        headings = re.findall(r"^#{1,3} (.+)$", text, re.M)
   412	        for req in REQUIRED_HEADINGS.get(fm.get("type"), []):
   413	            if not any(h.strip().startswith(req) for h in headings):
   414	                errors.append(f"{rel}: missing required heading `{req}` ({fm.get('type')} typedef §Required sections)")
   415	
   416	        # 8. no numbered-decision references (review-record §Commitment)
   417	        for i, line in enumerate(text.splitlines(), 1):
   418	            if DECISION_REF.search(line):
   419	                errors.append(f"{rel}:{i}: numbered-decision reference (decisions live as changes in the artifacts they affect)")
   420	
   421	        # 6. banned vocabulary
   422	        if rel.name != "README.md" and rel.name != "base-writing-style.md":
   423	            for i, line in enumerate(text.splitlines(), 1):
   424	                if "Replaces" in line or line.lstrip().startswith(("approved:", "ratified:")):
   425	                    continue
   426	                for term in BANNED:
   427	                    if term in line.lower():
   428	                        errors.append(f"{rel}:{i}: banned term `{term}` (use-defined-terms; see glossary)")
   429	
   430	        # 3. $ref sources in process data blocks
   431	        if fm.get("type") == "process-definition":
   432	            spec = yaml_blocks(text)
   433	            data = spec.get("data", {})
   434	            refs = collect_refs(data)
   435	            sourced = {}
   436	            def walk(node):
   437	                if isinstance(node, dict):
   438	                    if "$ref" in node and node.get("from"):
   439	                        sourced[node["$ref"]] = node["from"]
   440	                    for v in node.values():
   441	                        walk(v)
   442	                elif isinstance(node, list):
   443	                    for i in node:
   444	                        walk(i)
   445	            walk(data)
   446	            for ref in sorted(refs):
   447	                src = sourced.get(ref)
   448	                if not src:
   449	                    errors.append(f"{rel}: $ref `{ref}` has no `from:` source (process-definition §Data)")
   450	                elif src.startswith("pkg:"):
   451	                    if not PKG_RE.match(src):
   452	                        errors.append(f"{rel}: `{src}` is not pkg:<package>/<type> (process-definition §Data)")
   453	                else:
   454	                    target = (path.parent / src).resolve()
   455	                    if not target.exists():
   456	                        errors.append(f"{rel}: from `{src}` does not exist")
   457	                    else:
   458	                        tfm, _ = front_matter(target)
   459	                        if not tfm or tfm.get("defines") != ref:
   460	                            errors.append(f"{rel}: from `{src}` does not define `{ref}`")
   461	    errors += lint_requests()
   462	    errors += lint_briefs()
   463	    for path in sorted(PROCESSES.glob("*.md")):
   464	        fm, _ = front_matter(path)
   465	        if fm is not None and fm.get("type") == "process-definition":
   466	            errors += lint_process_tools(path)
   467	    errors += lint_guidance()
   468	    return errors

   471	def lint_requests():
   472	    """9. Each received request in requests/ carries the frontmatter of a
   473	    received ask (request typedef §Required frontmatter). The directory
   474	    may not exist yet: no requests, no violations."""
   475	    errors = []
   476	    if not REQUESTS.is_dir():
   477	        return errors
   478	    clause = "(request typedef §Required frontmatter)"
   479	    for path in sorted(REQUESTS.glob("*.md")):
   480	        rel = path.relative_to(BASIS.parent)
   481	        fm, _ = front_matter(path)
   482	        if fm is None:
   483	            errors.append(f"{rel}: front-matter missing or unparseable {clause}")
   484	            continue
   485	        for key in REQUEST_KEYS:
   486	            if key not in fm:
   487	                errors.append(f"{rel}: front-matter lacks `{key}` {clause}")
   488	        if "type" in fm and fm["type"] != "request":
   489	            errors.append(f"{rel}: `type` is `{fm['type']}`, not `request` {clause}")
   490	        if "status" in fm and fm["status"] not in REQUEST_STATUS:
   491	            errors.append(f"{rel}: `status` `{fm['status']}` not in {sorted(REQUEST_STATUS)} {clause}")
   492	        if "route" in fm and fm["route"] not in REQUEST_ROUTE:
   493	            errors.append(f"{rel}: `route` `{fm['route']}` not in {sorted(REQUEST_ROUTE)} {clause}")
   494	        target = fm.get("routed-to")
   495	        if target:
   496	            if not isinstance(target, str):
   497	                errors.append(f"{rel}: `routed-to` is not a link {clause}")
   498	            else:
   499	                # Resolve the part before any `#` fragment, as check 4 does
   500	                # for markdown links; the small-change lane's result is the
   501	                # request's own Result section by fragment, so the request's
   502	                # own repository-root path is accepted.
   503	                link = target.split("#", 1)[0]
   504	                if not link.startswith(("http://", "https://", "pkg:")) \
   505	                        and link != rel.as_posix() \
   506	                        and not (path.parent / link).exists():
   507	                    errors.append(f"{rel}: broken link `{target}` in `routed-to` {clause}")
   508	    return errors

   511	def lint_briefs():
   512	    """10. Each decision brief in briefs/ carries the decision-brief
   513	    typedef's closed field set and nothing outside it, and every
   514	    relates-to path resolves from the repository root. The directory may
   515	    not exist yet: no briefs, no violations. A brief is a file whose type
   516	    is decision-brief; the annexes beside it — a brief's linked full
   517	    material, of no fixed frontmatter — are not briefs and are not
   518	    walked."""
   519	    errors = []
   520	    if not BRIEFS.is_dir():
   521	        return errors
   522	    for path in sorted(BRIEFS.glob("*.md")):
   523	        fm, _ = front_matter(path)
   524	        if fm is not None and fm.get("type") == "decision-brief":
   525	            errors += lint_brief(path)
   526	    return errors

   529	def lint_brief(path):
   530	    """Check one decision brief by check 10's rules (decision-brief
   531	    typedef §Required frontmatter); the same rules for the tree walk
   532	    and for --brief. Paths under relates-to resolve from the repository
   533	    root, wherever the brief file lives."""
   534	    errors = []
   535	    clause = "(decision-brief typedef §Required frontmatter)"
   536	    root = BASIS.parent
   537	    full = path.resolve()
   538	    rel = full.relative_to(root) if full.is_relative_to(root) else path
   539	    fm, _ = front_matter(path)
   540	    if fm is None:
   541	        errors.append(f"{rel}: front-matter missing or unparseable {clause}")
   542	        return errors
   543	    for key in BRIEF_KEYS:
   544	        if key not in fm:
   545	            errors.append(f"{rel}: front-matter lacks `{key}` {clause}")
   546	    for key in fm:
   547	        if key not in BRIEF_KEYS and key not in BRIEF_OPTIONAL:
   548	            errors.append(f"{rel}: unknown front-matter key `{key}` — the field set is closed {clause}")
   549	    if "type" in fm and fm["type"] != "decision-brief":
   550	        errors.append(f"{rel}: `type` is `{fm['type']}`, not `decision-brief` {clause}")
   551	    if "status" in fm and fm["status"] not in BRIEF_STATUS:
   552	        errors.append(f"{rel}: `status` `{fm['status']}` not in {sorted(BRIEF_STATUS)} {clause}")
   553	    if "relates-to" in fm:
   554	        targets = fm["relates-to"]
   555	        if not isinstance(targets, list) or not targets:
   556	            errors.append(f"{rel}: `relates-to` is not a list of one or more paths {clause}")
   557	        else:
   558	            for target in targets:
   559	                if not isinstance(target, str) or not target:
   560	                    errors.append(f"{rel}: `relates-to` entry `{target}` is not a path {clause}")
   561	                elif not (root / target).is_file():
   562	                    errors.append(f"{rel}: `relates-to` path `{target}` does not resolve from the repository root {clause}")
   563	    return errors

   566	def lint_process_tools(path):
   567	    """11. Every repository tool path one process definition names exists
   568	    (process-definition typedef §Commitment: a process is not approved
   569	    while a step names a tool the repository lacks — the tool is a
   570	    request, routed before the process runs, never built mid-process).
   571	    A tool path is `basis/tools/<name>.py`, written in a step's `run:`
   572	    template or held as the `initial:` value of a data value or
   573	    parameter a template interpolates; it resolves from the repository
   574	    root. `bd`, `python3`, and the sh utilities are the environment and
   575	    are not checked. The same rule for the tree walk and for --process."""
   576	    errors = []
   577	    clause = "(process-definition typedef §Commitment)"
   578	    root = BASIS.parent
   579	    full = path.resolve()
   580	    rel = full.relative_to(root) if full.is_relative_to(root) else path
   581	    fm, text = front_matter(path)
   582	    if fm is None:
   583	        errors.append(f"{rel}: front-matter missing or unparseable {clause}")
   584	        return errors
   585	    spec = yaml_blocks(text)
   586	    sites = []  # (where the path is named, text to scan)
   587	    for block in ("data", "parameters"):
   588	        values = spec.get(block)
   589	        if isinstance(values, dict):
   590	            for name, decl in values.items():
   591	                if isinstance(decl, dict) and isinstance(decl.get("initial"), str):
   592	                    sites.append((f"`{block}.{name}` initial", decl["initial"]))
   593	    for step in spec.get("steps") or []:
   594	        if isinstance(step, dict) and isinstance(step.get("run"), str):
   595	            sites.append((f"step `{step.get('id', '?')}` run", step["run"]))
   596	    seen = set()
   597	    for where, scanned in sites:
   598	        for tool in TOOL_PATH.findall(scanned):
   599	            if (where, tool) in seen:
   600	                continue
   601	            seen.add((where, tool))
   602	            if not (root / tool).is_file():
   603	                errors.append(f"{rel}: {where} names `{tool}`, which does not exist in the repository {clause}")
   604	    return errors

   607	def lint_guidance():
   608	    """12. Each implementation guidance record in guidance/ at the
   609	    repository root carries the implementation-guidance typedef's
   610	    required frontmatter (implementation-guidance typedef §Required
   611	    frontmatter); each missing key is reported by name. The directory
   612	    may not exist yet: no records, no violations."""
   613	    errors = []
   614	    if not GUIDANCE.is_dir():
   615	        return errors
   616	    clause = "(implementation-guidance typedef §Required frontmatter)"
   617	    for path in sorted(GUIDANCE.glob("*.md")):
   618	        rel = path.relative_to(BASIS.parent)
   619	        fm, _ = front_matter(path)
   620	        if fm is None:
   621	            errors.append(f"{rel}: front-matter missing or unparseable {clause}")
   622	            continue
   623	        for key in GUIDANCE_KEYS:
   624	            if key not in fm:
   625	                errors.append(f"{rel}: front-matter lacks `{key}` {clause}")
   626	        if "type" in fm and fm["type"] != "implementation-guidance":
   627	            errors.append(f"{rel}: `type` is `{fm['type']}`, not `implementation-guidance` {clause}")
   628	    return errors

   631	def derive_chain(artifact_type):
   632	    """definition-chain is derived from document references, never authored:
   633	    typedef by `defines`, guideline and fitness by `target-type`, process by
   634	    `produces`, roles from the process steps, skill by `derived-from`."""
   635	    chain = {"artifact_type": artifact_type, "typedef": "", "guideline": "",
   636	             "fitness": "", "process": "", "roles": [], "skill": "", "status": "draft"}
   637	    docs = []
   638	    process_id = None
   639	    for path in sorted(BASIS.rglob("*.md")):
   640	        fm, text = front_matter(path)
   641	        if not fm:
   642	            continue
   643	        t = fm.get("type")
   644	        if t == "artifact-typedef" and fm.get("defines") == artifact_type:
   645	            chain["typedef"] = fm["id"]; docs.append(fm)
   646	        elif t == "quality-guideline" and fm.get("target-type") == artifact_type:
   647	            chain["guideline"] = fm["id"]; docs.append(fm)
   648	        elif t == "fitness-set" and fm.get("target-type") == artifact_type:
   649	            chain["fitness"] = fm["id"]; docs.append(fm)
   650	        elif t == "process-definition" and artifact_type in (fm.get("produces") or []):
   651	            chain["process"] = fm["id"]; process_id = fm["id"]; docs.append(fm)
   652	            spec = yaml_blocks(text)
   653	            roles = {s["run-by"]["role"] for s in spec.get("steps", [])
   654	                     if s.get("run-by", {}).get("role")}
   655	            chain["roles"] = sorted(roles)
   656	    if process_id:
   657	        for path in (BASIS.parent / ".claude" / "skills").rglob("SKILL.md"):
   658	            fm, _ = front_matter(path)
   659	            if fm and fm.get("derived-from") == process_id:
   660	                chain["skill"] = fm["id"]; docs.append(fm)
   661	    if docs and all(d.get("status") == "approved" for d in docs) and \
   662	       all(chain[k] for k in ("typedef", "guideline", "fitness", "process", "skill")):
   663	        chain["status"] = "approved"
   664	    return chain

   671	def fail(code, message):
   672	    """A failure the answer names: its code beside the message, one line on
   673	    standard error, the exit status the answer states for that code."""
   674	    print(f"{code}: {message}", file=sys.stderr)
   675	    sys.exit({"usage": 2, "unreadable": 2}[code])

   678	def readable(arg):
   679	    path = pathlib.Path(arg)
   680	    if not path.is_file():
   681	        fail("unreadable", f"{arg}: no such file")
   682	    try:
   683	        path.read_bytes()
   684	    except OSError as exc:
   685	        fail("unreadable", f"{arg}: cannot be read: {exc}")
   686	    return path

   689	def main():
   690	    args = sys.argv[1:]
   691	    # The standard question, answered before any other action; the other
   692	    # arguments are ignored (adr-2026-09-07-tool-answer §2).
   693	    if "--describe" in args:
   694	        sys.stdout.write(json.dumps(DESCRIPTION, indent=2) + "\n")
   695	        sys.exit(0)
   696	    if len(args) == 2 and args[0] == "--derive-chain":
   697	        print(yaml.safe_dump(derive_chain(args[1]), sort_keys=False).rstrip())
   698	        return
   699	    if len(args) == 2 and args[0] == "--brief":
   700	        errors = lint_brief(readable(args[1]))
   701	    elif len(args) == 2 and args[0] == "--process":
   702	        errors = lint_process_tools(readable(args[1]))
   703	    elif not args:
   704	        errors = lint()
   705	    else:
   706	        fail("usage", USAGE)
   707	    for e in errors:
   708	        print(e)
   709	    print(f"{'FAIL' if errors else 'PASS'}: {len(errors)} violation(s)")
   710	    sys.exit(1 if errors else 0)
